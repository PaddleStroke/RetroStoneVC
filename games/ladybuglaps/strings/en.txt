# Ladybug Laps: the game's own strings (docs/i18n.md). The kit's words (PRESS A TO %s, GAME OVER, PAUSED, GO!, 1ST...)
# are in games/common/strings. The game's name and the bots' names (RUBY, SUNNY, BLUEBELL, BASIL) are never translated.

#. The title prompt's verb: PRESS A TO <VERB> (the kit's PRESS_TO). max 12
VERB = RACE
#. The lap counter at the top left during a race: this lap, of how many. max 18
LAP = LAP %d/%d
#. Big letters (2x: two columns each) in the middle of the screen when you (1 player) or the leader start the last lap. max 36
FINAL_LAP = FINAL LAP!
#. Big letters (2x) when the race is over (you finished, or the leader). max 36
FINISH = FINISH!
#. 2-4 players: a racer (%s: a player tag P2, or a bot's name) fell too far behind the leader and left the screen: out of the race. max 34
BEHIND_P = %s LEFT BEHIND!
#. 2-4 players: a racer (%s: P2, or a bot's name) fell off the table: out of the race. max 34
FELL_P = %s FELL OFF!
#. Before the start: which race of the championship (or the cup) this is. max 24
RACE_N = RACE %d/%d
#. The tracks' names (before the start and on the results). max 22
TRACK_TEA = TEA TIME
#. max 22
TRACK_DESK = CRAYON DESK
#. max 22
TRACK_PICNIC = PICNIC
#. max 22
TRACK_VEG = VEGGIE PATCH
#. max 22
TRACK_PATIO = RAINY PATIO
#. A track driven the other way round; %s is its name. max 34
REVERSED = %s - REVERSED
#. The banner (2x letters on a panel that grows with them) of the results after each race. max 36
RESULTS = RESULTS
#. The blinking line under the results: A starts the next race. max 22
NEXT = A: NEXT RACE
#. The same after the last race, or when you are out. max 22
GO_ON = A: GO ON
#. 1 player, under the results: a 4th place ends the championship. max 34
OUT = 4TH PLACE: OUT!
#. 2-4 players: in the standings at the top right, in place of the points of a racer out of the race. max 3
OUT_TAG = OUT
#. The note at the foot of the game-over panel when you won all the races. max 22
CHAMPION = CHAMPION!
#. The note at the foot of the game-over panel: how far you got. max 22
REACHED = RACE %d OF %d
#. A hint at the bottom of the screen at the start of the first race (1 player): the d-pad, A and B. max 38
HINT = {}: STEER - A: GO - B: BRAKE
